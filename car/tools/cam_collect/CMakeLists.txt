cmake_minimum_required(VERSION 3.10)
project(UcrCollectCam LANGUAGES CXX)

if(NOT UNIX)
    message(FATAL_ERROR "ucr_collect_cam can Linux/POSIX de gioi han thoi gian camera")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgcodecs videoio imgproc OPTIONAL_COMPONENTS highgui)
find_package(Threads REQUIRED)
set(_cam_common "${CMAKE_CURRENT_SOURCE_DIR}/../collect_common")
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE _cam_git OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 1)
if(NOT _cam_git)
    set(_cam_git "unknown")
endif()
add_executable(ucr_collect_cam main.cpp "${_cam_common}/pass_store.cpp" "${_cam_common}/preview.cpp")
target_include_directories(ucr_collect_cam PRIVATE "${_cam_common}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ucr_collect_cam PRIVATE ${OpenCV_LIBS} Threads::Threads)
target_compile_definitions(ucr_collect_cam PRIVATE UCR_CAM_GIT_REVISION="${_cam_git}")
if(OpenCV_highgui_FOUND)
    target_compile_definitions(ucr_collect_cam PRIVATE UCR_DRIVE_HAVE_HIGHGUI=1)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ucr_collect_cam PRIVATE stdc++fs)
endif()

include(CTest)
if(BUILD_TESTING)
    # Bản chỉ để test: có hook nguồn giả; bản xe ucr_collect_cam không chứa hook.
    add_executable(ucr_collect_cam_hooks main.cpp "${_cam_common}/pass_store.cpp" "${_cam_common}/preview.cpp")
    target_compile_definitions(ucr_collect_cam_hooks PRIVATE UCR_CAM_TEST_HOOKS=1 UCR_DRIVE_TEST_HOOKS=1 UCR_CAM_GIT_REVISION="${_cam_git}")
    target_include_directories(ucr_collect_cam_hooks PRIVATE "${_cam_common}" ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(ucr_collect_cam_hooks PRIVATE ${OpenCV_LIBS} Threads::Threads)
    if(OpenCV_highgui_FOUND)
        target_compile_definitions(ucr_collect_cam_hooks PRIVATE UCR_DRIVE_HAVE_HIGHGUI=1)
    endif()
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(ucr_collect_cam_hooks PRIVATE stdc++fs)
    endif()
    add_executable(test_cam_collect_options test_options.cpp)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(test_cam_collect_options PRIVATE stdc++fs)
    endif()
    add_test(NAME cam_collect_options COMMAND test_cam_collect_options)
    set_tests_properties(cam_collect_options PROPERTIES TIMEOUT 120)
    find_package(PythonInterp 3.6 REQUIRED)
    find_program(CAM_COLLECT_NM nm)
    find_program(CAM_COLLECT_LDD ldd)
    find_program(CAM_COLLECT_STRINGS strings)
    if(NOT CAM_COLLECT_NM OR NOT CAM_COLLECT_LDD OR NOT CAM_COLLECT_STRINGS)
        message(FATAL_ERROR "Can nm, ldd va strings de kiem tra binary khong lien ket ma thiet bi")
    endif()
    set(CAM_COLLECT_PASS_STORE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../collect_common/pass_store.cpp"
        CACHE FILEPATH "Nguon PassStore dung chung cho ca format")
    set(CAM_COLLECT_PASS_STORE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../collect_common"
        CACHE PATH "Thu muc header PassStore dung chung cho ca format")
    foreach(scenario frames signal_int signal_term signal_hup disk disk_off invalid pipeline pipeline_frames eof corrupt existing
                     symbols hooks_ignored defaults out_root session max_seconds start_delay no_frame_timeout stall_timeout warmup help
                     default_off pass_toggle pass_two off_guard escape esc_no_pass preview_ctrl_c
                     max_frames_across_passes format H1 H2 H3 M1 M2
                     preview_no_display preview_fail_start preview_die_off preview_close_off)
        set(CAM_COLLECT_TEST_BINARY $<TARGET_FILE:ucr_collect_cam>)
        if(scenario STREQUAL "no_frame_timeout" OR scenario STREQUAL "stall_timeout" OR scenario STREQUAL "warmup")
            set(CAM_COLLECT_TEST_BINARY $<TARGET_FILE:ucr_collect_cam_hooks>)
        endif()
        if(scenario STREQUAL "preview_fail_start" OR scenario STREQUAL "preview_die_off" OR scenario STREQUAL "preview_close_off")
            set(CAM_COLLECT_TEST_BINARY $<TARGET_FILE:ucr_collect_cam_hooks>)
        endif()
        add_test(NAME cam_collect_${scenario} COMMAND ${PYTHON_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/test_cam_collect.py ${CAM_COLLECT_TEST_BINARY}
            ${scenario} ${CAM_COLLECT_NM} ${CAM_COLLECT_LDD} ${CAM_COLLECT_STRINGS}
            $<TARGET_FILE:ucr_collect_cam_hooks>
            "${CAM_COLLECT_PASS_STORE_SOURCE}" "${CAM_COLLECT_PASS_STORE_INCLUDE}"
            "${CMAKE_COMMAND}" "${CMAKE_CXX_COMPILER}")
        set_tests_properties(cam_collect_${scenario} PROPERTIES TIMEOUT 120 LABELS "acceptance")
        if(scenario MATCHES "^H[123]$")
            set_tests_properties(cam_collect_${scenario} PROPERTIES LABELS "acceptance;harness")
        endif()
        if(scenario MATCHES "^M[12]$")
            # Dựng bản lỗi -j1 tối đa 360 s, cộng baseline/probe và dọn tiến trình.
            set_tests_properties(cam_collect_${scenario} PROPERTIES TIMEOUT 600 LABELS "acceptance")
        endif()
    endforeach()
endif()
