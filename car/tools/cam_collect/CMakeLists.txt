cmake_minimum_required(VERSION 3.10)
project(UcrCollectCam LANGUAGES CXX)

if(NOT UNIX)
    message(FATAL_ERROR "ucr_collect_cam can Linux/POSIX de gioi han thoi gian camera")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgcodecs videoio imgproc OPTIONAL_COMPONENTS highgui)
find_package(Threads REQUIRED)
set(_cam_common "${CMAKE_CURRENT_SOURCE_DIR}/../collect_common")
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE _cam_git OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 1)
if(NOT _cam_git)
    set(_cam_git "unknown")
endif()
add_executable(ucr_collect_cam main.cpp "${_cam_common}/pass_store.cpp" "${_cam_common}/preview.cpp")
target_include_directories(ucr_collect_cam PRIVATE "${_cam_common}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ucr_collect_cam PRIVATE ${OpenCV_LIBS} Threads::Threads)
target_compile_definitions(ucr_collect_cam PRIVATE UCR_CAM_GIT_REVISION="${_cam_git}")
if(OpenCV_highgui_FOUND)
    target_compile_definitions(ucr_collect_cam PRIVATE UCR_DRIVE_HAVE_HIGHGUI=1)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ucr_collect_cam PRIVATE stdc++fs)
endif()

include(CTest)
