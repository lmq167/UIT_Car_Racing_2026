cmake_minimum_required(VERSION 3.10)
project(UcrCollectCam LANGUAGES CXX)

if(NOT UNIX)
    message(FATAL_ERROR "ucr_collect_cam can Linux/POSIX de gioi han thoi gian camera")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgcodecs videoio)
add_executable(ucr_collect_cam main.cpp)
target_include_directories(ucr_collect_cam PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ucr_collect_cam PRIVATE ${OpenCV_LIBS})
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ucr_collect_cam PRIVATE stdc++fs)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_cam_collect_options test_options.cpp)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(test_cam_collect_options PRIVATE stdc++fs)
    endif()
    add_test(NAME cam_collect_options COMMAND test_cam_collect_options)
    set_tests_properties(cam_collect_options PROPERTIES TIMEOUT 10)
    find_package(PythonInterp 3.6 REQUIRED)
    find_program(CAM_COLLECT_NM nm)
    if(NOT CAM_COLLECT_NM)
        message(FATAL_ERROR "Can nm de kiem tra binary khong lien ket ma thiet bi")
    endif()
    foreach(scenario frames signal_int signal_term disk invalid pipeline eof corrupt existing symbols)
        add_test(NAME cam_collect_${scenario} COMMAND ${PYTHON_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/test_cam_collect.py $<TARGET_FILE:ucr_collect_cam>
            ${scenario} ${CAM_COLLECT_NM})
        set_tests_properties(cam_collect_${scenario} PROPERTIES TIMEOUT 25)
    endforeach()
endif()
