if(NOT UNIX)
    message(FATAL_ERROR "ucr_auto cần Linux/POSIX")
endif()
find_package(Threads REQUIRED)
get_filename_component(_auto_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${_auto_root}"
    OUTPUT_VARIABLE _auto_git OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 1)
execute_process(COMMAND git rev-parse --abbrev-ref HEAD WORKING_DIRECTORY "${_auto_root}"
    OUTPUT_VARIABLE _auto_branch OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 1)
if(NOT _auto_git)
    set(_auto_git "unknown")
endif()
if(NOT _auto_branch)
    set(_auto_branch "unknown")
endif()
set(_auto_sources main.cpp log_writer.cpp ../collect_common/preview.cpp)
function(configure_auto target)
    target_include_directories(${target} PRIVATE "${_auto_root}/include" ${OpenCV_INCLUDE_DIRS})
    target_compile_definitions(${target} PRIVATE UCR_AUTO_PREVIEW=1
        UCR_AUTO_GIT_HASH="${_auto_git}" UCR_AUTO_GIT_BRANCH="${_auto_branch}")
    target_link_libraries(${target} PRIVATE ucr_control perception_cpu ${OpenCV_LIBS} Threads::Threads)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endfunction()
if(UCR_WITH_TRT)
    add_executable(ucr_auto ${_auto_sources}
        "${_auto_root}/src/hardware/Motor.cpp" "${_auto_root}/src/hardware/Servo.cpp"
        "${_auto_root}/src/sensor/BNO055.cpp" "${_auto_root}/src/sensor/TOF.cpp"
        "${_auto_root}/src/common/Logger.cpp" "${_auto_root}/src/common/TelemetryClient.cpp")
    configure_auto(ucr_auto)
    target_link_libraries(ucr_auto PRIVATE perception_trt "${TENSORRT_LIBRARY}")
    target_include_directories(ucr_auto PRIVATE "${TENSORRT_INCLUDE_DIR}" "${UCR_CUDA_INCLUDE_DIR}")
    if(OpenCV_highgui_FOUND)
        target_compile_definitions(ucr_auto PRIVATE UCR_DRIVE_HAVE_HIGHGUI=1)
    endif()
    if(UCR_FAKE_HW)
        target_include_directories(ucr_auto BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../collect_common/stub_i2c")
        target_sources(ucr_auto PRIVATE ../collect_common/stub_i2c/smbus.cpp)
    else()
        target_link_libraries(ucr_auto PRIVATE tof_backend i2c)
    endif()
else()
    message(STATUS "Bỏ qua ucr_auto: UCR_WITH_TRT=OFF; chỉ hooks dùng engine đường giả dưới BUILD_TESTING")
endif()
if(BUILD_TESTING)
    add_executable(ucr_auto_hooks ${_auto_sources} hooks.cpp)
    configure_auto(ucr_auto_hooks)
    target_compile_definitions(ucr_auto_hooks PRIVATE UCR_AUTO_TEST_HOOKS=1)
    include(${CMAKE_CURRENT_LIST_DIR}/tests.cmake)
    add_executable(test_debug_view test_debug_view.cpp)
    target_include_directories(test_debug_view PRIVATE "${_auto_root}/include" ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(test_debug_view PRIVATE ucr_control ${OpenCV_LIBS} Threads::Threads)
    add_test(NAME auto_debug_view COMMAND test_debug_view)
    set_tests_properties(auto_debug_view PROPERTIES LABELS "unit" TIMEOUT 30)
endif()
