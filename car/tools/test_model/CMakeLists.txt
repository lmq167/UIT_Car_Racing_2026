cmake_minimum_required(VERSION 3.10)
project(UcrTestModel LANGUAGES CXX)
if(NOT UNIX)
    message(FATAL_ERROR "ucr_test_model can Linux/POSIX")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4.1 QUIET COMPONENTS core imgcodecs videoio imgproc dnn OPTIONAL_COMPONENTS highgui)
# Thiếu dnn chỉ bỏ đích này, không được làm hỏng cấu hình chương trình chính trên xe.
if(NOT OpenCV_FOUND)
    message(WARNING "OpenCV khong co module dnn: bo qua ucr_test_model")
    return()
endif()
find_package(Threads REQUIRED)
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE _git OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 1)
if(NOT _git)
    set(_git "unknown")
endif()
# Thư viện hiển thị model: KHÔNG liên kết Motor/Servo/Camera (test nm -C kiểm).
add_library(test_model_view STATIC model_view/ModelInfo.cpp model_view/DnnMaskModel.cpp
    model_view/MaskRender.cpp model_view/ModelWorker.cpp)
target_include_directories(test_model_view PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(test_model_view PUBLIC ${OpenCV_LIBS} Threads::Threads)

add_executable(ucr_test_model main.cpp ../collect_common/preview.cpp ../collect_common/pass_store.cpp
    ${_root}/src/hardware/Motor.cpp ${_root}/src/hardware/Servo.cpp ${_root}/src/common/Logger.cpp)
target_include_directories(ucr_test_model PRIVATE ${_root}/include)
target_link_libraries(ucr_test_model PRIVATE test_model_view)
if(UCR_FAKE_HW)
    target_include_directories(ucr_test_model BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../collect_common/stub_i2c)
    target_sources(ucr_test_model PRIVATE ../collect_common/stub_i2c/smbus.cpp)
else()
    target_link_libraries(ucr_test_model PRIVATE i2c)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ucr_test_model PRIVATE -ffunction-sections -fdata-sections)
    set_property(TARGET ucr_test_model APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--gc-sections,--strip-debug")
endif()
if(OpenCV_highgui_FOUND)
    target_compile_definitions(ucr_test_model PRIVATE UCR_DRIVE_HAVE_HIGHGUI=1)
endif()
set_target_properties(ucr_test_model PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_definitions(ucr_test_model PRIVATE UCR_DRIVE_GIT_REVISION="${_git}")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ucr_test_model PRIVATE stdc++fs)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(ucr_test_model_hooks main.cpp ../collect_common/preview.cpp ../collect_common/pass_store.cpp)
    target_compile_definitions(ucr_test_model_hooks PRIVATE UCR_DRIVE_TEST_HOOKS=1 UCR_DRIVE_GIT_REVISION="${_git}")
    target_link_libraries(ucr_test_model_hooks PRIVATE test_model_view)
    set_target_properties(ucr_test_model_hooks PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(ucr_test_model_hooks PRIVATE stdc++fs)
    endif()
    include(${CMAKE_CURRENT_LIST_DIR}/tests.cmake)
endif()
