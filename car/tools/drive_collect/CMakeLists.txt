cmake_minimum_required(VERSION 3.10)
project(UcrDriveCollect LANGUAGES CXX)
if(NOT UNIX)
    message(FATAL_ERROR "ucr_drive_collect can Linux/POSIX de dieu khien terminal va camera")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgcodecs videoio imgproc OPTIONAL_COMPONENTS highgui)
find_package(Threads REQUIRED)
get_filename_component(UCR_DRIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(ucr_drive_collect main.cpp ../collect_common/preview.cpp ../collect_common/pass_store.cpp
    ${UCR_DRIVE_ROOT}/src/hardware/Motor.cpp
    ${UCR_DRIVE_ROOT}/src/hardware/Servo.cpp
    ${UCR_DRIVE_ROOT}/src/common/Logger.cpp)
target_include_directories(ucr_drive_collect PRIVATE ${UCR_DRIVE_ROOT}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ucr_drive_collect PRIVATE ${OpenCV_LIBS} Threads::Threads)
if(OpenCV_highgui_FOUND)
    target_compile_definitions(ucr_drive_collect PRIVATE UCR_DRIVE_HAVE_HIGHGUI=1)
endif()
if(UCR_FAKE_HW)
    # Chỉ thay lời gọi I²C khi kiểm thử; bản xe luôn dùng thư viện thật.
    target_include_directories(ucr_drive_collect BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/stub_i2c)
    target_sources(ucr_drive_collect PRIVATE stub_i2c/smbus.cpp)
else()
    target_link_libraries(ucr_drive_collect PRIVATE i2c)
endif()
# Loại các hàm phần cứng không dùng khỏi tệp nhị phân cuối.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ucr_drive_collect PRIVATE -ffunction-sections -fdata-sections)
    set_property(TARGET ucr_drive_collect APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--gc-sections,--strip-debug")
endif()
set_target_properties(ucr_drive_collect PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ucr_drive_collect PRIVATE stdc++fs)
endif()

execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE _drive_git OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 1)
if(NOT _drive_git)
    set(_drive_git "unknown")
endif()
target_compile_definitions(ucr_drive_collect PRIVATE UCR_DRIVE_GIT_REVISION="${_drive_git}")
include(CTest)
if(BUILD_TESTING)
    # Bản giả không mở UART/I2C; không truyền định nghĩa giả sang bản xe.
    add_executable(ucr_drive_collect_hooks main.cpp ../collect_common/preview.cpp ../collect_common/pass_store.cpp)
    set_target_properties(ucr_drive_collect_hooks PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_compile_definitions(ucr_drive_collect_hooks PRIVATE UCR_DRIVE_TEST_HOOKS=1 UCR_DRIVE_GIT_REVISION="${_drive_git}")
    target_include_directories(ucr_drive_collect_hooks PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(ucr_drive_collect_hooks PRIVATE ${OpenCV_LIBS} Threads::Threads)
    add_executable(test_drive_collect_options test_options.cpp)
    target_include_directories(test_drive_collect_options PRIVATE ${OpenCV_INCLUDE_DIRS})
    add_executable(test_drive_collect_reverse test_reverse.cpp)
    target_include_directories(test_drive_collect_reverse PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(test_drive_collect_reverse PRIVATE Threads::Threads)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(ucr_drive_collect_hooks PRIVATE stdc++fs)
        target_link_libraries(test_drive_collect_options PRIVATE stdc++fs)
        target_link_libraries(test_drive_collect_reverse PRIVATE stdc++fs)
    endif()
    add_test(NAME drive_collect_options COMMAND test_drive_collect_options)
    set_tests_properties(drive_collect_options PROPERTIES TIMEOUT 15)
    add_test(NAME drive_collect_reverse_contract COMMAND test_drive_collect_reverse)
    set_tests_properties(drive_collect_reverse_contract PROPERTIES TIMEOUT 15)
    find_package(PythonInterp 3.6 REQUIRED)
    add_test(NAME drive_collect_startup
        COMMAND "${PYTHON_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/test_startup.py"
        --hooks "$<TARGET_FILE:ucr_drive_collect_hooks>")
    set_tests_properties(drive_collect_startup PROPERTIES TIMEOUT 60)
    add_executable(test_preview_protocol ../collect_common/test_preview_protocol.cpp ../collect_common/preview.cpp)
    target_compile_definitions(test_preview_protocol PRIVATE UCR_DRIVE_TEST_HOOKS=1)
    target_include_directories(test_preview_protocol PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(test_preview_protocol PRIVATE ${OpenCV_LIBS} Threads::Threads)
    add_test(NAME drive_collect_preview_protocol COMMAND test_preview_protocol)
    set_tests_properties(drive_collect_preview_protocol PROPERTIES TIMEOUT 30)
    include(${CMAKE_CURRENT_LIST_DIR}/../collect_common/pass_store_tests.cmake)
    include(${CMAKE_CURRENT_LIST_DIR}/tests.cmake OPTIONAL)
    include(${CMAKE_CURRENT_LIST_DIR}/../collect_common/tests_preview.cmake OPTIONAL)
endif()
