find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgproc imgcodecs dnn)
add_library(perception_cpu STATIC ../../src/perception/InfoJson.cpp ../../src/perception/RoadSegmenter.cpp
    ../../src/perception/YoloDetector.cpp)
target_include_directories(perception_cpu PUBLIC ../../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(perception_cpu PUBLIC ${OpenCV_LIBS})

if(BUILD_TESTING)

    target_link_libraries(test_yolo PRIVATE perception_cpu)

if(UCR_WITH_TRT)
    set(TENSORRT_LIB_DIR "/usr/lib/aarch64-linux-gnu" CACHE PATH "Thư mục thư viện TensorRT")
    set(CUDA_TOOLKIT_ROOT_DIR "/usr/local/cuda-10.2" CACHE PATH "Thư mục CUDA 10.2")
    find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATHS /usr/include/aarch64-linux-gnu)
    find_library(TENSORRT_LIBRARY nvinfer PATHS "${TENSORRT_LIB_DIR}" NO_DEFAULT_PATH)
    find_path(UCR_CUDA_INCLUDE_DIR cuda_runtime_api.h PATHS "${CUDA_TOOLKIT_ROOT_DIR}/include" NO_DEFAULT_PATH)
    find_library(UCR_CUDART_LIBRARY cudart PATHS "${CUDA_TOOLKIT_ROOT_DIR}/lib64" NO_DEFAULT_PATH)
    if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY OR NOT UCR_CUDA_INCLUDE_DIR OR NOT UCR_CUDART_LIBRARY)
        message(FATAL_ERROR "Thiếu TensorRT/CUDA. Đặt TENSORRT_INCLUDE_DIR, TENSORRT_LIB_DIR và CUDA_TOOLKIT_ROOT_DIR (CUDA 10.2).")
    endif()
    add_library(perception_trt STATIC ../../src/perception/TrtEngine.cpp)
    target_compile_definitions(perception_trt PUBLIC UCR_WITH_TRT=1)
    target_include_directories(perception_trt PRIVATE "${TENSORRT_INCLUDE_DIR}" "${UCR_CUDA_INCLUDE_DIR}")
    target_link_libraries(perception_trt PUBLIC perception_cpu PRIVATE "${TENSORRT_LIBRARY}" "${UCR_CUDART_LIBRARY}")
    foreach(tool trt_bench check_engine)
        add_executable(${tool} ${tool}.cpp)
        target_link_libraries(${tool} PRIVATE perception_trt)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
            target_link_libraries(${tool} PRIVATE stdc++fs)
        endif()
    endforeach()
endif()
