cmake_minimum_required(VERSION 3.10)
project(UcrCollectDrive LANGUAGES CXX)
if(NOT UNIX)
    message(FATAL_ERROR "ucr_collect_drive can Linux/POSIX")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgcodecs videoio imgproc OPTIONAL_COMPONENTS highgui)
find_package(Threads REQUIRED)
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE _git OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 1)
if(NOT _git)
    set(_git "unknown")
endif()
add_executable(ucr_collect_drive main.cpp ../collect_common/preview.cpp ../collect_common/pass_store.cpp
    ${_root}/src/hardware/Motor.cpp ${_root}/src/hardware/Servo.cpp ${_root}/src/common/Logger.cpp)
target_include_directories(ucr_collect_drive PRIVATE ${_root}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ucr_collect_drive PRIVATE ${OpenCV_LIBS} Threads::Threads)
if(UCR_FAKE_HW)
    target_include_directories(ucr_collect_drive BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../drive_collect/stub_i2c)
    target_sources(ucr_collect_drive PRIVATE ../drive_collect/stub_i2c/smbus.cpp)
else()
    target_link_libraries(ucr_collect_drive PRIVATE i2c)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ucr_collect_drive PRIVATE -ffunction-sections -fdata-sections)
    set_property(TARGET ucr_collect_drive APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--gc-sections,--strip-debug")
endif()
if(OpenCV_highgui_FOUND)
    target_compile_definitions(ucr_collect_drive PRIVATE UCR_DRIVE_HAVE_HIGHGUI=1)
endif()
set_target_properties(ucr_collect_drive PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_definitions(ucr_collect_drive PRIVATE UCR_DRIVE_GIT_REVISION="${_git}")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(ucr_collect_drive PRIVATE stdc++fs)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(ucr_collect_drive_hooks main.cpp ../collect_common/preview.cpp ../collect_common/pass_store.cpp)
    target_compile_definitions(ucr_collect_drive_hooks PRIVATE UCR_DRIVE_TEST_HOOKS=1 UCR_DRIVE_GIT_REVISION="${_git}")
    target_include_directories(ucr_collect_drive_hooks PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(ucr_collect_drive_hooks PRIVATE ${OpenCV_LIBS} Threads::Threads)
    set_target_properties(ucr_collect_drive_hooks PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
        target_link_libraries(ucr_collect_drive_hooks PRIVATE stdc++fs)
    endif()
    include(${CMAKE_CURRENT_LIST_DIR}/tests.cmake OPTIONAL)
endif()
