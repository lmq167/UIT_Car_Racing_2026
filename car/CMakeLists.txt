cmake_minimum_required(VERSION 3.10)
project(UCR2026 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(UCR_FAKE_HW "Dung thiet bi gia lap" OFF)
option(UCR_WITH_TRT "Bật công cụ nhận thức TensorRT" OFF)
find_package(Threads REQUIRED)
find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui)
# Dấu vân tay chỉ gồm mã chạy trên xe, không đọc bộ kiểm thử độc lập.
file(GLOB_RECURSE UCR_SOURCE_FILES "src/*.cpp" "src/*.h" "include/*.h" "scripts/*")
list(APPEND UCR_SOURCE_FILES "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/src/CMakeLists.txt")
list(SORT UCR_SOURCE_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${UCR_SOURCE_FILES})
set(UCR_SOURCE_HASHES "")
foreach(source ${UCR_SOURCE_FILES})
    file(SHA256 "${source}" digest)
    string(APPEND UCR_SOURCE_HASHES "${digest}")
endforeach()
string(SHA256 UCR_SOURCE_ID "${UCR_SOURCE_HASHES}")
add_subdirectory(src)
add_subdirectory(tools/cam_collect)
add_subdirectory(tools/collect_drive)
add_subdirectory(tools/test_model)
add_executable(main main.cpp)
target_compile_definitions(main PRIVATE UCR_ENV_CHECK_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/scripts/env_check.sh")
target_link_libraries(main PRIVATE app_core)
add_executable(ucr_test_camera tests/hardware/test_camera.cpp src/hardware/Camera.cpp)
target_include_directories(ucr_test_camera PRIVATE include tests/hardware ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ucr_test_camera PRIVATE ${OpenCV_LIBS})
include(CTest)
add_subdirectory(tools/perception)
add_library(ucr_control STATIC src/control/ControlParams.cpp src/control/LaneController.cpp src/control/Planner.cpp src/control/ParkScan.cpp)
target_include_directories(ucr_control PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ucr_control PUBLIC opencv_core)
add_subdirectory(tools/auto_drive)
add_subdirectory(tools/replay_control)
if(BUILD_TESTING)
    add_subdirectory(tools/traj_eval)
    add_subdirectory(tests/acceptance)
endif()
