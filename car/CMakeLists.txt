cmake_minimum_required(VERSION 3.10)
project(UCR2026 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(UCR_FAKE_HW "Dung thiet bi gia lap" OFF)
find_package(Threads REQUIRED)
find_package(OpenCV 4.1 REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui)
add_subdirectory(src)
add_executable(main main.cpp)
target_link_libraries(main PRIVATE app_core)
add_executable(ucr_stop stop.cpp)
target_link_libraries(ucr_stop PRIVATE app_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_safety tests/test_safety.cpp)
    target_link_libraries(test_safety PRIVATE app_core)
    add_test(NAME safety COMMAND test_safety)
    add_test(NAME run_wrapper COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_run_wrapper.sh)
endif()
